sha256_file() {
    shasum -a 256 "$1" | awk 'NR == 1 { print $1 }'
}

sha256_blob() {
    git show "$1:$2" | shasum -a 256 | awk 'NR == 1 { print $1 }'
}

sha256_driver_blob() {
    driver_commit="$1"
    driver_begin="$2"
    driver_end="$3"
    git show "$driver_commit:$plan_path" |
        awk \
            -v begin="$driver_begin" \
            -v end="$driver_end" '
BEGIN { bc=0; ec=0; inside=0; bad=0; n=0 }
$0 == begin {
    bc++
    if (bc != 1 || ec != 0 || inside != 0) bad=1
    inside=1
    next
}
$0 == end {
    ec++
    if (bc != 1 || ec != 1 || inside != 1) bad=1
    inside=0
    next
}
inside { body[++n]=$0 }
END {
    if (bad || bc != 1 || ec != 1 || inside != 0 || n == 0) exit 73
    for (i=1; i<=n; i++) print body[i]
}' |
        shasum -a 256 | awk 'NR == 1 { print $1 }'
}

file_bytes() {

file_lines() {

require_exact_output() {
    local expected_output="$1"
    local actual_output
    shift
    actual_output="$("$@")" || return 1
    test "$actual_output" = "$expected_output" || return 1
}

require_regular_sha() {
    test -f "$1" || return 1
    test ! -L "$1" || return 1
    require_exact_output "$2" sha256_file "$1" || return 1
}

require_absent() {
    test ! -e "$1" || return 1
    test ! -L "$1" || return 1
}

require_empty_index() {
    require_exact_output '' git diff --cached --name-only || return 1
}

require_no_nonignored_untracked() {
    require_exact_output '' git ls-files --others --exclude-standard || return 1
}

require_git_clean() {
    require_empty_index || return 1
    require_exact_output '' git diff --name-only || return 1
    require_no_nonignored_untracked || return 1
    require_exact_output '' git status --short || return 1
    git diff --check || return 1
}

require_exact_commit() {
    local actual_scope
    commit="$1"
    parent="$2"
    subject="$3"
    expected_scope="$4"
    require_exact_output commit git cat-file -t "$commit" || return 1
    require_exact_output "$parent" git show -s --format='%P' "$commit" || return 1
    require_exact_output "$subject" git show -s --format='%s' "$commit" || return 1
    actual_scope="$(git diff-tree --no-commit-id --name-only -r "$commit" |
        LC_ALL=C sort)" || return 1
    test "$actual_scope" = "$expected_scope" || return 1
}

}

prefix_sha256() {
    dd if="$1" bs=1 count="$2" 2>/dev/null |
        shasum -a 256 | awk 'NR == 1 { print $1 }'
}

reject_failure_infrastructure() {
    local diagnostic_scan_status

    else
        diagnostic_scan_status="$?"

    test "$diagnostic_scan_status" = 1 || return 1
}

emit_expected_suffix() {

verify_ledgers() {
    local expected_suffix_base64
    local progress_suffix_base64
    local report_size
    local report_suffix_base64
    local progress_size
    local suffix_rows
    local terminal_byte
    suffix_rows="$1"
    require_ledger_prefixes || return 1
    expected_suffix_base64="$(emit_expected_suffix "$suffix_rows" |
        /usr/bin/base64)" || return 1
    report_suffix_base64="$(dd if="$report_path" bs=1 \
        skip="$report_anchor_bytes" 2>/dev/null | /usr/bin/base64)" || return 1
    progress_suffix_base64="$(dd if="$progress_path" bs=1 \
        skip="$progress_anchor_bytes" 2>/dev/null | /usr/bin/base64)" || return 1
    test "$report_suffix_base64" = "$expected_suffix_base64" || return 1
    test "$progress_suffix_base64" = "$expected_suffix_base64" || return 1
    test "$report_suffix_base64" = "$progress_suffix_base64" || return 1
    if [ "$suffix_rows" = 0 ]; then
        require_exact_output "$report_anchor_bytes" \
            file_bytes "$report_path" || return 1
        require_exact_output "$progress_anchor_bytes" \
            file_bytes "$progress_path" || return 1
    else
        report_size="$(file_bytes "$report_path")" || return 1
        progress_size="$(file_bytes "$progress_path")" || return 1
        test "$report_size" -gt "$report_anchor_bytes" || return 1
        test "$progress_size" -gt "$progress_anchor_bytes" || return 1
        terminal_byte="$(dd if="$report_path" bs=1 skip=$((report_size - 1)) \
            count=1 2>/dev/null | od -An -tu1 | tr -d '[:space:]')" || return 1
        test "$terminal_byte" = 10 || return 1
        terminal_byte="$(dd if="$progress_path" bs=1 skip=$((progress_size - 1)) \
            count=1 2>/dev/null | od -An -tu1 | tr -d '[:space:]')" || return 1
        test "$terminal_byte" = 10 || return 1
    fi
    return 0
}

append_row_pair() {
    before_rows="$1"
    key="$2"
    value="$3"
    verify_ledgers "$before_rows" || return 1
    require_exact_output 0 key_count "$report_path" "$key" || return 1

    printf '%s %s: %s\n' "$row_prefix" "$key" "$value" >> "$progress_path" || return 1
    verify_ledgers "$((before_rows + 1))" || return 1
    require_pair_value "$key" "$value" || return 1
}

cleanup_patch_snapshot() {
    snapshot_cleanup_failed=0
    if [ -n "$patch_snapshot" ] &&
       { [ -e "$patch_snapshot" ] || [ -L "$patch_snapshot" ]; }; then
        if [ -f "$patch_snapshot" ] && [ ! -L "$patch_snapshot" ]; then
            /bin/rm -f "$patch_snapshot" || snapshot_cleanup_failed=1
        else
            snapshot_cleanup_failed=1
        fi
    fi
    if [ -n "$patch_snapshot_dir" ] &&
       { [ -e "$patch_snapshot_dir" ] || [ -L "$patch_snapshot_dir" ]; }; then
        if [ -d "$patch_snapshot_dir" ] && [ ! -L "$patch_snapshot_dir" ]; then
            /bin/rmdir "$patch_snapshot_dir" || snapshot_cleanup_failed=1
        else
            snapshot_cleanup_failed=1
        fi
    fi
    if [ "$snapshot_cleanup_failed" = 0 ]; then
        patch_snapshot=''
        patch_snapshot_dir=''
        return 0
    fi
    return 1
}

cleanup_patch_on_exit() {
    cleanup_status="$?"
    trap - EXIT
    trap '' HUP INT TERM
    set +e
    cleanup_failed=0
    inverse_completed=0
    if [ "$patch_applied" = 1 ]; then
        (
            set -euo pipefail
            require_regular_sha "$patch_snapshot" "$patch_sha"
        )
        snapshot_auth_status="$?"
        if [ "$snapshot_auth_status" = 0 ]; then
            git apply -R --check "$patch_snapshot"
            inverse_check_status="$?"
            if [ "$inverse_check_status" = 0 ]; then
                git apply -R "$patch_snapshot"
                inverse_status="$?"
                if [ "$inverse_status" != 0 ]; then
                    cleanup_failed=1
                else
                    inverse_completed=1
                fi
            else
                (
                    set -euo pipefail
                    require_role_baseline "$evidence_role"
                )
                already_restored_status="$?"
                if [ "$already_restored_status" = 0 ]; then
                    inverse_completed=1
                else
                    cleanup_failed=1
                fi
            fi
        else
            cleanup_failed=1
        fi
    fi
    (
        set -euo pipefail
        require_role_baseline "$evidence_role"
    )
    restored_baseline_status="$?"
    if [ "$restored_baseline_status" != 0 ]; then
        cleanup_failed=1
    elif [ "$patch_applied" = 1 ] && [ "$inverse_completed" = 1 ]; then
        patch_applied=0
    fi
    if [ "$cleanup_failed" = 0 ]; then
        cleanup_patch_snapshot
        snapshot_cleanup_status="$?"
        if [ "$snapshot_cleanup_status" != 0 ]; then
            cleanup_failed=1
        fi
    fi
    if [ "$cleanup_failed" != 0 ]; then
        printf '%s\n' \
            "Task 4.2 correction r03 cleanup failed; preserve snapshot directory: $patch_snapshot_dir" >&2
        if [ -f "$patch_snapshot" ] && [ ! -L "$patch_snapshot" ]; then
            preserved_snapshot_sha="$(sha256_file "$patch_snapshot")"
            preserved_snapshot_sha_status="$?"
            if [ "$preserved_snapshot_sha_status" = 0 ]; then
                printf '%s\n' \
                    "Task 4.2 correction r03 preserved patch snapshot sha256: $preserved_snapshot_sha" >&2
            else
                printf '%s\n' \
                    'Task 4.2 correction r03 preserved patch snapshot sha256 unavailable' >&2
            fi
        fi
        exit 126
    fi
    exit "$cleanup_status"
}

        /private/tmp/task-4.2-correction-r03-patch.XXXXXX)" || return 1
    test -d "$patch_snapshot_dir" || return 1
    test ! -L "$patch_snapshot_dir" || return 1
    patch_snapshot="$patch_snapshot_dir/patch"
    trap cleanup_patch_on_exit EXIT

    trap 'exit 130' INT
    trap 'exit 143' TERM
    /bin/cp "$patch_path" "$patch_snapshot" || return 1
    require_regular_sha "$patch_snapshot" "$patch_sha" || return 1

    patch_applied=1

    require_mutated_state "$evidence_role" "$target_path" || return 1
    set +e
    set -o noclobber
    (
        set -euo pipefail
        printf '%s\n' "$marker" || exit 1
        "$dotnet_bin" build "$unit_project" \
            --configuration Release --no-restore --nologo \
            -warnaserror -t:Rebuild -m:1 -p:UseSharedCompilation=false || exit "$?"
        "$dotnet_bin" run --project "$unit_project" \
            --configuration Release --no-restore --no-build -- \
            --cohort driver-initial || exit "$?"
    ) > "$evidence_output" 2>&1
    captured_status="$?"

    status_write_status="$?"
    set +o noclobber
    set -e
    require_regular_sha "$patch_path" "$patch_sha" || return 1
    require_regular_sha "$patch_snapshot" "$patch_sha" || return 1
    trap '' HUP INT TERM
    git apply -R --check "$patch_snapshot" || return 1
    git apply -R "$patch_snapshot" || return 1
    patch_applied=0
    trap 'exit 129' HUP
    trap 'exit 130' INT
    trap 'exit 143' TERM
    test "$status_write_status" = 0 || return 1
    require_role_baseline "$evidence_role" || return 1
    set +e
    cleanup_patch_snapshot
    normal_snapshot_cleanup_status="$?"
    set -e
    if [ "$normal_snapshot_cleanup_status" != 0 ]; then
        exit 126
    fi
    trap - EXIT HUP INT TERM
    run_focused_green || return 1
    validate_failure_pair "$evidence_role" "$mutation_id" || return 1
    evidence_record="$(evidence_value \
        "$evidence_role" "$mutation_id" "$restoration")" || return 1
    append_row_pair "$before_rows" \
        "$evidence_role M$mutation_id evidence" "$evidence_record" || return 1
    verify_manifest "$((before_pairs + 1))" || return 1
}
