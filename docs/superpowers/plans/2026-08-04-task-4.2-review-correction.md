actual_round_paths() {
    for candidate in "$artifact_dir"/*; do
        if [ ! -e "$candidate" ] && [ ! -L "$candidate" ]; then
            continue
        fi
        candidate_name="$(basename "$candidate")"
        case "$candidate_name" in
            task-4.2-correction-r01-*|task-4.2-review-correction-r01-brief.md)
                printf '%s\n' "$candidate"
                ;;
        esac
    done | LC_ALL=C sort
}

verify_manifest() {
    evidence_pairs="$1"
    test "$evidence_pairs" -ge 0
    test "$evidence_pairs" -le 54
    materialized=$((26 + 2 * evidence_pairs))
    test "$(manifest_paths | wc -l | tr -d '[:space:]')" = 134
    ordinal=0
    while IFS= read -r artifact_path; do
        ordinal=$((ordinal + 1))
        if [ "$ordinal" -le "$materialized" ]; then
            test -f "$artifact_path"
            test ! -L "$artifact_path"
        else
            require_absent "$artifact_path"
        fi
    done < <(manifest_paths)
    test "$ordinal" = 134
    cmp -s \
        <(actual_round_paths) \
        <(manifest_paths | sed -n "1,${materialized}p" | LC_ALL=C sort)
}

prefix_sha256() {
    dd if="$1" bs=1 count="$2" 2>/dev/null |
        shasum -a 256 | awk 'NR == 1 { print $1 }'
}

require_ledger_prefixes() {
    test -f "$report_path"
    test ! -L "$report_path"
    test -f "$progress_path"
    test ! -L "$progress_path"
    ! test "$report_path" -ef "$progress_path"
    test "$(file_bytes "$report_path")" -ge "$report_anchor_bytes"
    test "$(file_bytes "$progress_path")" -ge "$progress_anchor_bytes"
    test "$(prefix_sha256 "$report_path" "$report_anchor_bytes")" = "$report_anchor_sha"
    test "$(prefix_sha256 "$progress_path" "$progress_anchor_bytes")" = "$progress_anchor_sha"
    test "$(dd if="$report_path" bs=1 skip=$((report_anchor_bytes - 1)) count=1 2>/dev/null | od -An -tu1 | tr -d '[:space:]')" = 10
    test "$(dd if="$progress_path" bs=1 skip=$((progress_anchor_bytes - 1)) count=1 2>/dev/null | od -An -tu1 | tr -d '[:space:]')" = 10
}

reject_failure_infrastructure() {
    evidence_output="$1"
    if awk '
        /: error / || /: warning / || /error CS/ || /warning CS/ ||
        /Build FAILED[.]/ || /NETSDK/ || /assets file/ || /SDK/ ||
        /MSB[0-9]/ || /permission denied/ || /Permission denied/ ||
        /Operation not permitted/ || /UnauthorizedAccessException/ ||
        /timed out/ || /timeout/ || /NU1/ || /package/ || /Package/ ||
        /restore/ || /No such file/ || /command not found/ { bad=1 }

    fi

validate_failure_pair() {
    evidence_role="$1"
    mutation_id="$2"
    mutation_meta "$mutation_id"
    evidence_paths "$evidence_role" "$mutation_id"

    test -f "$evidence_output"

    test "$(file_lines "$evidence_status")" = 1

    test "$(sha256_file "$evidence_status")" = "$status_one_sha"
    test "$(sed -n '1p' "$evidence_output")" = "$marker"
    test "$(awk '$0 == "Build succeeded." { n++ } END { print n+0 }' "$evidence_output")" = 1
    test "$(awk '/^[[:space:]]*0 Warning[(]s[)]$/ { n++ } END { print n+0 }' "$evidence_output")" = 1

        "$failure_prefix"*"$fragment"*) ;;

    esac

}

    fi

The task-local launchers below repeat the complete fail-closed extraction
sequence. The buffered extractor emits no driver bytes until it has observed
exactly one ordered marker pair and a nonempty body. The only invocation of
`task42_main` is appended after successful extraction. `pipefail` therefore
propagates an empty, truncated, failed, or signal-terminated producer, even if
the downstream shell could parse its prefix. Each launcher authenticates the
definition-only body, syntax-checks the exact combined stream, reauthenticates
the body immediately before execution, strips shell startup variables, and
gives the running driver `/dev/null` rather than its source stream as stdin.

This is the exact Task 1 launcher. Later tasks repeat it literally, changing
only the documented `HEAD` selector and final `run_driver` call arguments.
